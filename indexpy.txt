class Character:
    def __init__(self, health, position):
        self.health=health
        self.position=position

class Warrior(Character):
    def __init__(self, shield, position):
        super().__init__(health=100, position=position)

        self.health = 250

        self.shield = shield

class Magician(Character):
    def __init__(self, position, staff):
        super().__init__(health=250, position=position)
        self.health=150
        self.staff=staff

mercy = Magician(position="Rear", staff="Phoenix wand")
print(f"A magicians position is in the {mercy.position},and she has {mercy.health} hp and she has {mercy.staff}")
