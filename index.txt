class Player:
    def __init__(self, name, age, scores):
        self.name = name
        self.age = age
        self.scores = scores

    def result(self, score, name, game):
        self.scores[game] = score

    def calculate_average(self):
        total = sum(self.scores.values())
        number_of_games = len(self.scores)
        return total / number_of_games


maya = Player("Maya", 19, {
    "Chess": 85,
    "Tennis": 90,
    "Swimming": 78
})

print(maya.calculate_average())
